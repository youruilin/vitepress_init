吸顶searchBar组件的设计与实现 ​

考察了各大平台，包括pdd、咸鱼、豆瓣等top应用，他们在首页做了全局吸顶搜索bar，我也试着做了一个；

scrollNavBar.vue

vue
<script setup lang="ts">
import { ref, watch, onMounted, getCurrentInstance, computed, nextTick } from 'vue'

interface TabItem {
  key: string
  label: string
}

const props = defineProps<{
  tabs: TabItem[]
  activeKey: string | undefined
}>()

const emit = defineEmits<{
  (e: 'change', key: string): void
  (e: 'search'): void
}>()

// 新增：暴露给父组件的导航栏高度
const navbarHeight = ref(0)

const scrollLeft = ref(0)
const isMounted = ref(false)
const instance = getCurrentInstance()
const enableScroll = ref(true)
const touchMoved = ref(false)
const isAnimating = ref(false)
const scrollAnimationDuration = ref(300) // 动态动画时长

// 点击事件处理
const handleTabClick = (key: string) => {
  if (isAnimating.value) return
  console.log('🔥 Tab被点击了！key:', key)
  emit('change', key)
}

const handleSearchClick = () => {
  console.log('🔍 搜索框被点击')
  emit('search')
}

// 计算当前活跃tab索引
const activeIndex = computed(() => {
  return props.tabs.findIndex(tab => tab.key === props.activeKey)
})

// 计算目标tab距离当前活跃tab的间隔数
const getTabDistance = (targetKey: string): number => {
  const currentIndex = activeIndex.value
  const targetIndex = props.tabs.findIndex(tab => tab.key === targetKey)

  if (currentIndex === -1 || targetIndex === -1) return 0

  return Math.abs(targetIndex - currentIndex)
}

// 根据间隔数计算动画时长（核心优化）
const calculateAnimationDuration = (distance: number): number => {
  if (distance <= 0) return 300

  // 基础时长 + 距离系数
  const baseDuration = 100
  const distanceFactor = 80 // 每个标签增加的时间（毫秒）

  // 限制最大动画时长
  const maxDuration = 400
  const calculatedDuration = baseDuration + distance * distanceFactor

  return Math.min(calculatedDuration, maxDuration)
}

// 获取tab元素尺寸
const getTabRects = (): Promise<DOMRect[]> => {
  return new Promise<DOMRect[]>(resolve => {
    if (!instance) {
      resolve([])
      return
    }

    const query = uni.createSelectorQuery().in(instance)
    query.selectAll('.navbar-tab').boundingClientRect()
    query.exec(res => {
      resolve((res?.[0] as DOMRect[]) || [])
    })
  })
}

// 居中函数 - 修复版本
const centerActiveTab = async (key: string, immediate = false) => {
  if (!isMounted.value) {
    setTimeout(() => centerActiveTab(key, immediate), 100)
    return
  }

  if (isAnimating.value && !immediate) return
  isAnimating.value = true

  try {
    const tabRects = await getTabRects()
    if (!tabRects || tabRects.length === 0) return

    // 查找目标tab索引
    const targetIndex = props.tabs.findIndex(tab => tab.key === key)
    if (targetIndex === -1) return

    const targetRect = tabRects[targetIndex]
    if (!targetRect) return

    const screenWidth = uni.getSystemInfoSync().windowWidth

    // 计算目标tab左侧的总宽度
    let totalWidthBefore = 0
    for (let i = 0; i < targetIndex; i++) {
      if (tabRects[i]) {
        totalWidthBefore += tabRects[i]?.width || 0
      }
    }

    // 计算目标滚动位置（居中）
    const targetScrollLeft = totalWidthBefore - (screenWidth - targetRect.width) / 2
    const clampedScrollLeft = Math.max(0, targetScrollLeft)

    // 计算间隔距离和动画时长
    const distance = getTabDistance(key)
    const animationDuration = calculateAnimationDuration(distance)

    console.log(
      `🎯 Tab切换: ${props.activeKey} → ${key}, 间隔: ${distance}, 动画时长: ${animationDuration}ms`,
    )

    // 更新动画时长
    scrollAnimationDuration.value = animationDuration

    // 如果有明显的位置变化才执行动画
    if (Math.abs(scrollLeft.value - clampedScrollLeft) > 1) {
      scrollLeft.value = clampedScrollLeft

      // 动画结束后重置状态
      setTimeout(() => {
        isAnimating.value = false
      }, animationDuration + 50)
    } else {
      // 位置变化很小，直接完成
      scrollLeft.value = clampedScrollLeft
      isAnimating.value = false
    }
  } catch (error) {
    console.error('居中动画出错:', error)
    isAnimating.value = false
  }
}

// 新增：获取导航栏高度
// 新增：获取导航栏高度
const measureNavbarHeight = (): Promise<number> => {
  return new Promise<number>(resolve => {
    if (!instance) {
      resolve(0)
      return
    }

    // 使用 nextTick 确保 DOM 已渲染
    nextTick(() => {
      const query = uni.createSelectorQuery().in(instance)

      // 查询整个导航栏容器
      query.select('.navbar').boundingClientRect()
      query.exec(res => {
        if (res && res[0]) {
          // 类型断言确保安全访问
          const rect = res[0] as UniApp.NodeInfo
          console.log('📏 测量导航栏高度:', rect.height)
          navbarHeight.value = rect.height || 0
          resolve(rect.height || 0)
        } else {
          console.warn('❌ 测量导航栏高度失败')
          resolve(0)
        }
      })
    })
  })
}

// 监听activeKey变化
watch(
  () => props.activeKey,
  key => {
    if (key) {
      // 使用setTimeout确保DOM已更新
      setTimeout(() => {
        centerActiveTab(key)
      }, 100)
    }
  },
  { immediate: true },
)

// 组件挂载
onMounted(() => {
  console.log('✅ Navbar组件已挂载')

  setTimeout(async () => {
    isMounted.value = true
    console.log('🚀 组件就绪')

    // 初始化时居中当前激活的tab
    if (props.activeKey) {
      centerActiveTab(props.activeKey, true)
    }

    // 新增：测量并设置导航栏高度
    try {
      const height = await measureNavbarHeight()
      if (height > 0) {
        console.log('✅ 导航栏高度已获取:', height, 'px')
      }
    } catch (error) {
      console.error('获取导航栏高度失败:', error)
    }
  }, 300)
})

// 新增：暴露给父组件的属性和方法
defineExpose({
  navbarHeight,
})

// 触摸事件处理
const onTouchStart = () => {
  touchMoved.value = false
  enableScroll.value = true
}

const onTouchMove = () => {
  touchMoved.value = true
}

const onTouchEnd = (key: string) => {
  // 如果没有发生滑动，判定为点击
  if (!touchMoved.value && !isAnimating.value) {
    enableScroll.value = false
    handleTabClick(key)

    // 下一帧恢复滚动能力
    setTimeout(() => {
      enableScroll.value = true
    }, 0)
  }
}

// 获取屏幕边界到安全区域距离
const { safeAreaInsets } = uni.getWindowInfo()

const halfNavbarHeight = computed(() => {
  const top = safeAreaInsets?.top ?? 0
  return top / 2 + 'px'
})

const menuButtonInfo = uni.getMenuButtonBoundingClientRect()

// 动态样式类
const scrollViewClass = computed(() => {
  return ['navbar-scroll', isAnimating.value ? 'navbar-scroll-animating' : '']
})
</script>

<template>
  <view
    ref="navbarElement"
    class="navbar"
    style="position: fixed; top: 0; left: 0; right: 0; z-index: 999"
  >
    <!-- 安全区域占位 -->
    <!-- <view :style="{ height: menuButtonInfo.top + menuButtonInfo.height }"></view> -->

    <view
      class="navbar-wrapper"
      :style="{ paddingTop: menuButtonInfo.top + 'px' }"
    >
      <!-- 搜索框 -->
      <view
        class="search-box"
        :style="{ height: menuButtonInfo.height + 'px' }"
        @tap="handleSearchClick"
      >
        <text class="search-placeholder">搜索商品 / 补贴</text>
      </view>

      <!-- 使用动态的scroll-animation-duration -->
      <scroll-view
        :class="scrollViewClass"
        :scroll-x="enableScroll"
        :scroll-left="scrollLeft"
        scroll-with-animation
        :scroll-animation-duration="scrollAnimationDuration"
      >
        <view class="navbar-inner">
          <view
            v-for="item in tabs"
            :key="item.key"
            class="navbar-tab-wrapper"
            @touchstart="onTouchStart"
            @touchmove="onTouchMove"
            @touchend="onTouchEnd(item.key)"
          >
            <view
              class="navbar-tab"
              :class="{ active: item.key === activeKey }"
            >
              <view class="tab-content">
                <text class="tab-text">{{ item.label }}</text>
                <view
                  v-if="item.key === activeKey"
                  class="active-indicator"
                />
              </view>
            </view>
          </view>
        </view>
      </scroll-view>
    </view>
  </view>
</template>

<script setup lang="ts">
import { ref } from 'vue'
import ScrollNavbar from '@/components/ScrollNavbar.vue'
import TabSwiper from '@/components/TabSwiper.vue'

const tabs = [
  { key: 'recommend', label: '推荐' },
  { key: 'festival', label: '年货节' },
  { key: 'subsidy1', label: '政府补贴1' },
  { key: 'subsidy2', label: '政府补贴2' },
  { key: 'subsidy3', label: '政府补贴3' },
  { key: 'subsidy4', label: '政府补贴4' },
  { key: 'subsidy5', label: '政府补贴5' },
]

const activeIndex = ref(0)
const activeTab = ref(tabs[0]?.key)

/** ✅ navbar 点击 */
function onNavbarChange(key: string) {
  const index = tabs.findIndex(t => t.key === key)
  if (index !== -1) {
    activeIndex.value = index
    activeTab.value = key // 🔴 立刻高亮
  }
}

/** ✅ swiper 横滑 */
function onSwiperChange(index: number) {
  activeIndex.value = index
  activeTab.value = tabs[index]?.key
}

function onSearchClick() {
  console.log('跳转搜索页')
}
</script>

<template>
  <ScrollNavbar
    :tabs="tabs"
    :active-key="activeTab"
    @change="onNavbarChange"
    @search="onSearchClick"
  />

  <TabSwiper
    :tabs="tabs"
    v-model="activeIndex"
    @change="onSwiperChange"
  >
    <template #recommend>
      <RecommendTab />1
    </template>

    <template #festival>
      <FestivalTab />2
    </template>

    <template #subsidy>
      <SubsidyTab />3
    </template>
  </TabSwiper>
</template>
