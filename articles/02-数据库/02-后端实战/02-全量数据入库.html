# ================= 安全请求 =================
def safe_get(url, params=None):
    """安全请求，带重试机制"""
    for attempt in range(3):
        try:
            res = requests.get(url, params=params, headers=HEADERS, timeout=10)
            
            if res.status_code == 200:
                try:
                    return res.json()
                except:
                    return None
            else:
                if attempt < 2:
                    time.sleep(2)
                    continue
                return None
                
        except Exception as e:
            if attempt < 2:
                time.sleep(2)
                continue
            return None
    
    return None


# ================= 转换列式数据为行式数据 =================
def convert_column_to_row(column_data):
    """将列式存储的数据转换为行式存储"""
    if not column_data or not isinstance(column_data, dict):
        return []
    
    fields = list(column_data.keys())
    if not fields:
        return []
    
    first_field = fields[0]
    row_count = len(column_data[first_field])
    
    rows = []
    for i in range(row_count):
        row = {}
        for field in fields:
            value = column_data[field].get(str(i))
            row[field] = value
        rows.append(row)
    
    return rows

class BatchStorage:
    """分批存储管理器"""
    
    def __init__(self, data_dir="variety_batch_data"):
        self.data_dir = Path(data_dir)
        self.data_dir.mkdir(exist_ok=True)
        self.pages_per_batch = CONFIG["PAGES_PER_BATCH"]
        self.progress_file = self.data_dir / "progress.json"
        self.progress = self._load_progress()
    
    def _get_batch_name(self, page_num):
        batch_index = (page_num - 1) // self.pages_per_batch + 1
        start_page = (batch_index - 1) * self.pages_per_batch + 1
        end_page = min(batch_index * self.pages_per_batch, CONFIG["MAX_PAGES"])
        return f"batch_{batch_index:03d}_pages_{start_page:04d}-{end_page:04d}.json"
    
    def _get_batch_index(self, page_num):
        return (page_num - 1) // self.pages_per_batch + 1
    
    def _load_progress(self):
        if self.progress_file.exists():
            with open(self.progress_file, 'r', encoding='utf-8') as f:
                progress = json.load(f)
                print(f"📂 加载进度: 已爬取 {progress.get('total_records', 0):,} 条, 最后页码 {progress.get('last_page', 0)}")
                return progress
        
        return {
            "version": "3.0",
            "start_time": datetime.now().isoformat(),
            "last_page": 0,
            "total_records": 0,
            "crawled_pages": [],
            "batches": {}
        }
    
    def _save_progress(self):
        self.progress["last_update"] = datetime.now().isoformat()
        with open(self.progress_file, 'w', encoding='utf-8') as f:
            json.dump(self.progress, f, indent=2, ensure_ascii=False)
    
    def _load_batch_file(self, batch_file):
        file_path = self.data_dir / batch_file
        if not file_path.exists():
            return {}
        with open(file_path, 'r', encoding='utf-8') as f:
            return json.load(f)
    
    def _save_batch_file(self, batch_file, data):
        file_path = self.data_dir / batch_file
        with open(file_path, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        
        file_size = file_path.stat().st_size / (1024 * 1024)
        print(f"  💾 批次文件已更新: {batch_file} ({file_size:.2f}MB)")
    
    def save_page(self, page_num, page_data):
        if not page_data:
            return False
        
        batch_file = self._get_batch_name(page_num)
        batch_index = self._get_batch_index(page_num)
        
        batch_data = self._load_batch_file(batch_file)
        batch_data[str(page_num)] = page_data
        self._save_batch_file(batch_file, batch_data)
        
        if page_num not in self.progress["crawled_pages"]:
            self.progress["crawled_pages"].append(page_num)
        
        self.progress["last_page"] = page_num
        self.progress["total_records"] += len(page_data)
        
        if batch_index not in self.progress["batches"]:
            self.progress["batches"][batch_index] = {
                "file": batch_file,
                "pages": [],
                "records": 0,
                "start_page": (batch_index - 1) * self.pages_per_batch + 1,
                "end_page": min(batch_index * self.pages_per_batch, CONFIG["MAX_PAGES"])
            }
        
        batch_info = self.progress["batches"][batch_index]
        if page_num not in batch_info["pages"]:
            batch_info["pages"].append(page_num)
        batch_info["records"] += len(page_data)
        
        self._save_progress()
        
        print(f"  ✅ 第{page_num}页已存入: {batch_file}")
        print(f"  📊 本批次: {len(batch_info['pages'])}/{batch_info['end_page']-batch_info['start_page']+1}页, {batch_info['records']:,}/{CONFIG['RECORDS_PER_BATCH']:,}条")
        print(f"  📊 全局: {self.progress['total_records']:,}/{CONFIG['TOTAL_RECORDS']:,}条 ({self.progress['total_records']/CONFIG['TOTAL_RECORDS']*100:.1f}%)")
        
        return True
    
    def is_page_crawled(self, page_num):
        return page_num in self.progress["crawled_pages"]
    
    def get_next_page(self, start_page=1):
        for page in range(start_page, start_page + 10000):
            if not self.is_page_crawled(page):
                return page
        return None
    
    def get_statistics(self):
        total_batches = len(self.progress["batches"])
        total_pages = len(self.progress["crawled_pages"])
        total_records = self.progress["total_records"]
        
        total_size = 0
        for batch_info in self.progress["batches"].values():
            file_path = self.data_dir / batch_info["file"]
            if file_path.exists():
                total_size += file_path.stat().st_size
        
        completion_rate = (total_records / CONFIG["TOTAL_RECORDS"] * 100) if CONFIG["TOTAL_RECORDS"] > 0 else 0
        
        return {
            "total_batches": total_batches,
            "total_pages": total_pages,
            "total_records": total_records,
            "expected_records": CONFIG["TOTAL_RECORDS"],
            "completion_rate": completion_rate,
            "last_page": self.progress["last_page"],
            "total_size_mb": total_size / (1024 * 1024),
            "data_dir": str(self.data_dir),
        }
    
    def get_all_batches(self):
        return sorted(self.progress["batches"].keys())
    
    def get_all_data_generator(self):
        for batch_index in self.get_all_batches():
            batch_info = self.progress["batches"][batch_index]
            batch_data = self._load_batch_file(batch_info["file"])
            for page_num in sorted(batch_data.keys(), key=int):
                for record in batch_data[page_num]:
                    yield record
    
    def get_base_judgementnos(self):
        """获取基准审定编号（来自第一个批次文件）"""
        batch_files = sorted(self.data_dir.glob("batch_*.json"))
        if not batch_files:
            return set()
        
        first_batch_file = batch_files[0]
        print(f"📂 读取基准批次文件: {first_batch_file.name}")
        
        with open(first_batch_file, 'r', encoding='utf-8') as f:
            batch_data = json.load(f)
        
        judgementnos = set()
        for page_num, page_data in batch_data.items():
            for record in page_data:
                judgementno = record.get('judgementno')
                if judgementno:
                    judgementnos.add(judgementno)
        
        print(f"  基准审定编号数: {len(judgementnos):,}")
        return judgementnos

class IncrementalManager:
    """增量数据管理器 - 单独存储增量数据"""
    
    def __init__(self, inc_dir="incremental_data"):
        self.inc_dir = Path(inc_dir)
        self.inc_dir.mkdir(exist_ok=True)
        self.index_file = self.inc_dir / "incremental_index.json"
        self.index = self._load_index()
    
    def _load_index(self):
        """加载增量数据索引"""
        if self.index_file.exists():
            with open(self.index_file, 'r', encoding='utf-8') as f:
                index = json.load(f)
                print(f"📂 加载增量索引: {len(index.get('incremental_files', []))} 个增量文件")
                return index
        
        return {
            "version": "1.0",
            "created_at": datetime.now().isoformat(),
            "incremental_files": [],  # 增量文件列表
            "all_judgementnos": set(),  # 所有已存储的审定编号（包括主数据+所有增量）
            "total_pages": 0,
            "total_records": 0
        }
    
    def _save_index(self):
        """保存增量数据索引"""
        # 转换set为列表以便JSON序列化
        index_copy = self.index.copy()
        index_copy["all_judgementnos"] = list(self.index["all_judgementnos"])
        
        with open(self.index_file, 'w', encoding='utf-8') as f:
            json.dump(index_copy, f, indent=2, ensure_ascii=False)
    
    def load_all_known_judgementnos(self, base_storage):
        """
        加载所有已知的审定编号（主数据 + 所有增量数据）
        用于检测新数据
        """
        all_judgementnos = set()
        
        # 1. 加载主数据的审定编号
        print(f"\n🔍 加载主数据审定编号...")
        base_judgementnos = base_storage.get_base_judgementnos()
        all_judgementnos.update(base_judgementnos)
        print(f"  主数据: {len(base_judgementnos):,} 个")
        
        # 2. 加载所有增量数据的审定编号
        if self.index.get("all_judgementnos"):
            # 从索引加载（快速）
            inc_judgementnos = set(self.index.get("all_judgementnos", []))
            all_judgementnos.update(inc_judgementnos)
            print(f"  增量数据(索引): {len(inc_judgementnos):,} 个")
        else:
            # 从文件加载（首次使用）
            print(f"  正在加载增量数据文件...")
            for inc_file_info in self.index.get("incremental_files", []):
                file_path = self.inc_dir / inc_file_info["filename"]
                if file_path.exists():
                    with open(file_path, 'r', encoding='utf-8') as f:
                        inc_data = json.load(f)
                    
                    for page_num, page_data in inc_data.items():
                        for record in page_data:
                            judgementno = record.get('judgementno')
                            if judgementno:
                                all_judgementnos.add(judgementno)
                    
                    print(f"    {inc_file_info['filename']}: {inc_file_info['records']} 条")
        
        # 更新索引
        self.index["all_judgementnos"] = all_judgementnos
        self._save_index()
        
        print(f"\n📊 总计已知审定编号: {len(all_judgementnos):,} 个")
        return all_judgementnos
    
    def save_incremental_data(self, new_pages_data, new_page_numbers):
        """
        保存增量数据到单独文件
        new_pages_data: {page_num: page_data, ...}
        new_page_numbers: 新数据的页码列表
        """
        timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
        inc_file = self.inc_dir / f"incremental_{timestamp}.json"
        
        # 保存增量数据
        inc_data = {
            "crawl_time": datetime.now().isoformat(),
            "pages": new_page_numbers,
            "total_pages": len(new_page_numbers),
            "total_records": len(new_page_numbers) * CONFIG["PAGE_SIZE"],
            "data": new_pages_data
        }
        
        with open(inc_file, 'w', encoding='utf-8') as f:
            json.dump(inc_data, f, indent=2, ensure_ascii=False)
        
        # 更新索引
        self.index["incremental_files"].append({
            "filename": inc_file.name,
            "time": datetime.now().isoformat(),
            "pages": new_page_numbers,
            "records": len(new_page_numbers) * CONFIG["PAGE_SIZE"]
        })
        
        # 更新已知审定编号集合
        for page_num, page_data in new_pages_data.items():
            for record in page_data:
                judgementno = record.get('judgementno')
                if judgementno:
                    self.index["all_judgementnos"].add(judgementno)
        
        self.index["total_pages"] += len(new_page_numbers)
        self.index["total_records"] += len(new_page_numbers) * CONFIG["PAGE_SIZE"]
        self._save_index()
        
        file_size = inc_file.stat().st_size / (1024 * 1024)
        print(f"\n💾 增量数据已保存: {inc_file}")
        print(f"   文件大小: {file_size:.2f} MB")
        print(f"   新增页数: {len(new_page_numbers)}")
        print(f"   新增记录: {len(new_page_numbers) * CONFIG['PAGE_SIZE']:,} 条")
        
        return inc_file

# ================= 随机延迟函数 =================
def random_delay(page_num, total_pages, success_count):
    """随机延迟函数"""
    if page_num >= total_pages:
        return
    
    # 基础随机延迟 2-5 秒
    delay = random.uniform(CONFIG["MIN_INTERVAL"], CONFIG["MAX_INTERVAL"])
    
    # 每50页多休息一下
    if success_count > 0 and success_count % 50 == 0:
        rest_time = random.uniform(10, 20)
        print(f"\n  😴 已爬取 {success_count} 页，休息 {rest_time:.1f} 秒...")
        time.sleep(rest_time)
        return
    
    print(f"  ⏱️  等待 {delay:.2f} 秒后继续...")
    time.sleep(delay)

def crawl_batch(start_page=1, end_page=None):
    """分批爬取数据（带随机延迟）"""
    if end_page is None:
        end_page = CONFIG["MAX_PAGES"]
    
    storage = BatchStorage()
    
    print(f"\n{'='*70}")
    print(f"🚀 开始分批爬取任务（随机间隔模式）")
    print(f"{'='*70}")
    print(f"  配置信息:")
    print(f"    每页数量: {CONFIG['PAGE_SIZE']} 条")
    print(f"    每批页数: {CONFIG['PAGES_PER_BATCH']} 页")
    print(f"    每批记录: {CONFIG['RECORDS_PER_BATCH']:,} 条")
    print(f"  随机延迟:")
    print(f"    请求间隔: {CONFIG['MIN_INTERVAL']}-{CONFIG['MAX_INTERVAL']} 秒")
    print(f"    批次休息: 每50页休息10-20秒")
    print(f"  任务范围:")
    print(f"    起始页码: {start_page}")
    print(f"    结束页码: {end_page}")
    print(f"    预计总条数: {(end_page - start_page + 1) * CONFIG['PAGE_SIZE']:,} 条")
    print(f"    存储目录: {storage.data_dir}")
    print(f"{'='*70}\n")
    
    start_time = time.time()
    
    success_pages = 0
    empty_pages = 0
    failed_pages = 0
    
    for page in range(start_page, end_page + 1):
        # 检查是否已爬取
        if storage.is_page_crawled(page):
            print(f"⏭️  第{page}页已存在，跳过")
            continue
        
        print(f"\n📄 [进度: {page}/{end_page}] 正在爬取第 {page} 页...")
        
        try:
            data = get_rowData(page=page, page_size=CONFIG["PAGE_SIZE"])
            
            if not data:
                print(f"  ⚠️ 第{page}页无数据")
                empty_pages += 1
                if empty_pages >= 5:
                    print(f"\n⚠️ 连续{empty_pages}页无数据，已到达数据末尾")
                    print(f"   实际总页数: {success_pages}")
                    break
                # 无数据也等待一下
                if page < end_page:
                    time.sleep(random.uniform(1, 3))
                continue
            
            # 有数据，重置空页计数
            empty_pages = 0
            
            # 保存到批次文件
            storage.save_page(page, data)
            success_pages += 1
            
            # 随机延迟
            if page < end_page:
                random_delay(page, end_page, success_pages)
            
        except KeyboardInterrupt:
            print(f"\n\n⚠️ 用户中断，已保存进度到第{page-1}页")
            print(f"下次运行可选择「继续上次」")
            break
        except Exception as e:
            print(f"  ❌ 第{page}页爬取失败: {e}")
            failed_pages += 1
            time.sleep(5)
            continue
    
    elapsed_time = time.time() - start_time
    stats = storage.get_statistics()
    
    print(f"\n{'='*70}")
    print(f"🎉 爬取任务完成！")
    print(f"{'='*70}")
    print(f"  成功页数: {success_pages}")
    print(f"  空页数: {empty_pages}")
    print(f"  失败页数: {failed_pages}")
    print(f"  总记录数: {stats['total_records']:,} 条")
    print(f"  总批次数: {stats['total_batches']} 个文件")
    print(f"  总文件大小: {stats['total_size_mb']:.1f} MB")
    print(f"  耗时: {elapsed_time/60:.1f} 分钟 ({elapsed_time:.1f}秒)")
    print(f"  平均速度: {stats['total_records']/elapsed_time:.1f} 条/秒")
    print(f"  存储位置: {storage.data_dir}")
    print(f"{'='*70}")

    print(f"  主数据记录数: {stats['total_records']:,} 条")
    print(f"  主数据页数: {stats['total_pages']} 页")
    
    # 加载所有已知的审定编号（主数据 + 已有增量数据）
    print(f"\n🔍 正在加载已知审定编号...")
    known_judgementnos = inc_manager.load_all_known_judgementnos(storage)
    
    # 检测新数据
    print(f"\n🔍 检测新数据（从第1页开始）...")
    
    new_pages_data = {}
    new_page_numbers = []
    max_check_pages = 500  # 最多检查500页
    
    for page in range(1, max_check_pages + 1):
        print(f"  检查第 {page} 页...", end=" ")
        
        data = get_rowData(page=page, page_size=CONFIG["PAGE_SIZE"])
        
        if not data:
            print("无数据，停止检测")
            break
        
        # 检查这一页是否包含已知的审定编号
        has_known = False
        new_in_page = 0
        
        for record in data:
            judgementno = record.get('judgementno')
            if judgementno:
                if judgementno in known_judgementnos:
                    has_known = True
                else:
                    new_in_page += 1
        
        if has_known:
            if new_in_page > 0:
                # 新旧混合（边界页，理论上不应该发生）
                print(f"⚠️ 混合页 (新:{new_in_page}, 旧:{len(data)-new_in_page})")
                # 仍然保存新数据部分
                new_pages_data[str(page)] = data
                new_page_numbers.append(page)
            else:
                # 全是旧数据，边界已到
                print(f"全是旧数据，边界已找到")
            break
        else:
            # 全是新数据
            print(f"✨ 全新数据 ({len(data)}条)")
            new_pages_data[str(page)] = data
            new_page_numbers.append(page)
    
    if not new_page_numbers:
        print(f"\n✅ 没有发现新数据！数据已是最新。")
        return
    
    print(f"\n📊 检测结果:")
    print(f"  新数据页数: {len(new_page_numbers)} 页")
    print(f"  新数据记录数: {len(new_page_numbers) * CONFIG['PAGE_SIZE']:,} 条")
    print(f"  页码范围: 第 {new_page_numbers[0]} - {new_page_numbers[-1]} 页")

        print("已取消")
        return

def import_batch(records, cursor, batch_size=500):
    """
    批量导入记录（使用 INSERT IGNORE，配合唯一索引）
    """
    table_name = get_table_name()
    sql = f"""
        INSERT IGNORE INTO {table_name} (
            cropid, cropname, varietyname, varietyannouncementid,
            judgementno, judgementyear, judgementregion, judgementregionid,
            certificateno, breedingcompany, applycompany,
            varietycharacter, varietysource, plantrequirment,
            judgementsuggestion, outputexpression,
            isTuiguangHas, isVarietyApplyHas, isXukeByVarietynameHas,
            isXukeByApplycompanynameHas, istransgenosis, status
        ) VALUES (
            %s, %s, %s, %s,
            %s, %s, %s, %s,
            %s, %s, %s,
            %s, %s, %s,
            %s, %s,
            %s, %s, %s,
            %s, %s, %s
        )
    """
    
    inserted = 0
    for i in range(0, len(records), batch_size):
        batch = records[i:i+batch_size]
        try:
            cursor.executemany(sql, batch)
            # rowcount 返回实际影响的行数（INSERT IGNORE 会跳过重复）
            inserted += cursor.rowcount
            print(f"    ✅ 已入库: {inserted}/{len(records)} 条")
        except Exception as e:
            print(f"    ❌ 批次入库失败: {e}")
            # 降级：逐条插入
            for record in batch:
                try:
                    cursor.execute(sql, record)
                    if cursor.rowcount > 0:
                        inserted += 1
                except Exception as e2:
                    print(f"      ❌ 单条失败: {record[3]}, 错误: {e2}")
    
    return inserted

def load_records_from_files(data_dir, max_records=None, load_all_batches=True):
    """
    从JSON文件加载记录
    max_records: 最大加载条数（用于测试）
    load_all_batches: 是否加载所有批次（False表示只加载最新的批次）
    """
    data_path = Path(data_dir)
    batch_files = sorted(data_path.glob("batch_*.json"))
    
    if not batch_files:
        print(f"❌ 未找到JSON文件: {data_dir}")
        return []
    
    # 如果不是加载所有批次，只取最后一个
    if not load_all_batches and len(batch_files) > 1:
        batch_files = batch_files[-1:]
        print(f"📂 增量模式: 只检查最新批次文件")
    
    all_records = []
    
    for batch_file in batch_files:
        print(f"  加载: {batch_file.name}")
        with open(batch_file, 'r', encoding='utf-8') as f:
            batch_data = json.load(f)
        
        # 展开所有记录（按页码排序）
        for page_num in sorted(batch_data.keys(), key=int):
            page_records = batch_data[page_num]
            for record in page_records:
                all_records.append(record)
                if max_records and len(all_records) >= max_records:
                    return all_records
    
    return all_records

def import_from_single_json(json_file="all_variety_data.json", max_records=None, incremental=False):
    """
    从单个合并的JSON文件导入
    max_records: 最大导入条数（None表示全部）
    incremental: 是否为增量导入
    """
    table_name = get_table_name()
    
    print(f"\n📂 正在加载文件: {json_file}")
    
    with open(json_file, 'r', encoding='utf-8') as f:
        records = json.load(f)
    
    if max_records:
        records = records[:max_records]
    
    print(f"📊 共 {len(records)} 条记录")
    
    conn = get_connection()
    cursor = conn.cursor()
    
    # 如果是测试表，确保表存在
    if USE_TEST_TABLE:
        create_test_table_if_not_exists(cursor)
    
    # 解析记录
    parsed_records = []
    skipped_no_id = 0
    for record in records:
        parsed = parse_record(record)
        if parsed[3]:  # varietyannouncementid
            parsed_records.append(parsed)
        else:
            skipped_no_id += 1
    
    print(f"有效记录: {len(parsed_records)} 条")
    if skipped_no_id > 0:
        print(f"跳过(无ID): {skipped_no_id} 条")
    
    # 增量或全量导入
    if incremental:
        existing_ids = load_existing_announcement_ids(cursor, table_name)
        print(f"{table_name} 中已有: {len(existing_ids):,} 条")
        inserted, new_count = import_incremental_batch(parsed_records, cursor, existing_ids)
        print(f"新增记录: {new_count} 条")
    else:
        inserted = import_batch(parsed_records, cursor)
    
    conn.commit()
    
    cursor.close()
    conn.close()
    
    print(f"\n✅ 导入完成: 成功 {inserted}/{len(parsed_records)} 条")
    
    return inserted

(async () => {
  const conn = await mysql.createConnection({
    host: "192.168.10.224",
    user: "yunfeng",
    password: "386898136Di!",
    database: "yunfeng"
  });

  // ==========================
  const [stateRows] = await conn.execute(
    "SELECT last_variety_id FROM etl_state WHERE id = 1"
  );

  const lastId = stateRows[0].last_variety_id;
  console.log("📍 上次同步ID:", lastId);

  // ==========================
  // 2. 增量读取（核心修改点）
  // ==========================
  const [rows] = await conn.execute(
    "SELECT * FROM variety_data WHERE id > ? ORDER BY id ASC",
    [lastId]
  );

  if (rows.length === 0) {
    console.log("📭 无新增数据，ETL结束");
    await conn.end();
    return;
  }

  console.log(`📦 本次新增数据量: ${rows.length}`);

  // ==========================
  // 3. 内存结构
  // ==========================
  const companyMap = new Map();
  const companySet = new Set();
  const relationSet = new Set();

  // ==========================
  // 4. 已有 company
  // ==========================
  const [companies] = await conn.execute("SELECT * FROM company");

  companies.forEach(c => {
    companyMap.set(c.company_name, c.company_id);
  });

  const newCompanies = [];

  // ==========================
  // 5. ETL 主逻辑
  // ==========================
  let maxId = lastId;

  for (const item of rows) {

    maxId = Math.max(maxId, item.id);

    const varietyId = item.varietyannouncementid;

    const parse = (str, type) => {
      if (!str) return;

      const list = str.split("、");

      for (const name of list) {
        const clean = name.trim();
        if (!clean) continue;

        // company
        if (!companyMap.has(clean) && !companySet.has(clean)) {
          companySet.add(clean);
          newCompanies.push(clean);
        }

        // relation
        relationSet.add(`${varietyId}|${clean}|${type}`);
      }
    };

    parse(item.breedingcompany, 1);
    parse(item.applycompany, 2);
  }

  // ==========================
  // 6. 插入 company
  // ==========================
  console.log(`🏢 新公司数量: ${newCompanies.length}`);

  for (let i = 0; i < newCompanies.length; i += BATCH_SIZE) {
    const batch = newCompanies.slice(i, i + BATCH_SIZE);
    const values = batch.map(name => [name]);

    await conn.query(
      "INSERT IGNORE INTO company (company_name) VALUES ?",
      [values]
    );
  }

  // ==========================
  // 7. 更新 companyMap
  // ==========================
  const [updatedCompanies] = await conn.execute("SELECT * FROM company");

  companyMap.clear();
  updatedCompanies.forEach(c => {
    companyMap.set(c.company_name, c.company_id);
  });

  // ==========================
  // 8. 插入 relation
  // ==========================
  console.log(`🔗 relation数量: ${relationSet.size}`);

  const relationBatch = [];

  for (const key of relationSet) {
    const [varietyId, name, type] = key.split("|");
    const companyId = companyMap.get(name);

    if (!companyId) continue;

    relationBatch.push([
      varietyId,
      companyId,
      Number(type)
    ]);
  }

  for (let i = 0; i < relationBatch.length; i += BATCH_SIZE) {
    const batch = relationBatch.slice(i, i + BATCH_SIZE);

    await conn.query(
      `INSERT IGNORE INTO variety_company 
       (varietyannouncementid, company_id, company_type) 
       VALUES ?`,
      [batch]
    );
  }

  // ==========================
  // 9. 回写游标（核心新增）
  // ==========================
  await conn.execute(
    "UPDATE etl_state SET last_variety_id = ? WHERE id = 1",
    [maxId]
  );

  console.log("📌 更新游标:", maxId);

  console.log("✅ ETL COMPLETE");

  await conn.end();

})().catch(error => {
  console.error("❌ ETL 失败:", error);
  process.exit(1);
});

-- 完整的数据填充流程

-- 1. 清空表
TRUNCATE TABLE transgene_variety;

-- 2. 插入所有转基因品种（使用 transgenic_status）
INSERT INTO transgene_variety (variety_dict_id, judgementno_with_transgene, transgene_type)
SELECT 
  id,
  primary_judgementno,
  '转基因'
FROM variety_dict
WHERE transgenic_status = 1
  AND status = 1;

-- 3. 添加 varietyname 字段（方便匹配，或者 JOIN 时获取）
-- 注意：如果表没有 varietyname 字段，需要添加
ALTER TABLE transgene_variety ADD COLUMN varietyname VARCHAR(200) AFTER variety_dict_id;

-- 4. 更新品种名称
UPDATE transgene_variety tv
JOIN variety_dict vd ON tv.variety_dict_id = vd.id
SET tv.varietyname = vd.varietyname;

-- 5. 建立关联（去后缀匹配）
UPDATE transgene_variety tv
LEFT JOIN variety_dict vd ON (
  -- 去掉末尾的字母后缀
  REGEXP_REPLACE(tv.varietyname, '(D|K|R|TK|ZL|GS|GX|KK)$', '') = vd.varietyname
  -- AND vd.transgenic_status = 0
  AND vd.transgenic_status IN (0, 2)  -- 🔥 修改这里：允许 0 或 2
  AND vd.status = 1
)
SET tv.parent_variety_id = vd.id
WHERE tv.parent_variety_id IS NULL;

FROM transgene_variety;
