

# 后端api-server，基于nodejs+express

## ① 本地创建 api-server（后期跑通再部署）

在你 Windows / 本地开发机：

```
npm init -y
```

## ② 安装依赖

```
npm config get registry -- 查看当前源
npm config set registry https://registry.npmmirror.com/  -- 切换淘宝源
npm config set registry https://registry.npmjs.org/  -- 切回官方源

npm install express mysql2 cors dotenv 
```

## 配置 Nginx 反向代理

**`server_name` 指的是 HTTP 请求中 `Host` 头的值 = 被解析的域名！**

```bash
# 创建配置文件
cat > /etc/nginx/sites-available/yunfeng-api << 'EOF'
server {
    listen 8090;
    server_name yunfeng-server.iooio.cn;  # 替换为你的域名或 IP

    location / {
        proxy_pass http://localhost:3000;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_cache_bypass $http_upgrade;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
    }
}
EOF

# 创建软连接
sudo ln -s \
/etc/nginx/sites-available/yunfeng-api \
/etc/nginx/sites-enabled/

sudo nginx -t

sudo systemctl reload nginx
```

```bash
# https 配置

cat > /etc/nginx/sites-available/yunfeng-api << 'EOF'
server {
    listen 8090 ssl http2;

    server_name yunfeng-server.iooio.cn;

    ssl_certificate     /etc/letsencrypt/live/yunfeng-server.iooio.cn/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/yunfeng-server.iooio.cn/privkey.pem;

    location / {
        proxy_pass http://127.0.0.1:3000;
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;
    }
}
EOF
```
