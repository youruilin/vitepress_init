HEADERS = {

# ================= 安全请求 =================
def safe_get(url, params=None):
    try:
        res = requests.get(url, params=params, headers=HEADERS, timeout=10)
        print(f"  状态码: {res.status_code}")

        try:
            return res.json()
        except Exception:
            print(f"  ❌ 非JSON返回: {res.text[:100]}")
            return None

    except Exception as e:
        print(f"  ❌ 请求失败: {e}")
        return None


# ================= 转换列式数据为行式数据 =================
def convert_column_to_row(column_data):
    """将列式存储的数据转换为行式存储"""
    if not column_data or not isinstance(column_data, dict):
        return []
    
    fields = list(column_data.keys())
    if not fields:
        return []
    
    first_field = fields[0]
    row_count = len(column_data[first_field])
    
    rows = []
    for i in range(row_count):
        row = {}
        for field in fields:
            value = column_data[field].get(str(i))
            row[field] = value
        rows.append(row)
    
    return rows

def get_rowData(page=1, page_size=50):
    """
    获取单页数据
    page: 页码
    page_size: 每页数量（最大50）
    """
    url = f"{BASE_URL}/v1/data-service/api/getDwdSdVarietyannouncementList"

    params = {
        "page": page,
        "pageSize": page_size,  # 现在使用50
        "cropid": "",
        "judgementyear": "",

        "varietyname": "",
        "judgementno": "",
        "companyName": "",
        "istransgenosis": "",
        "status": "",
        "introductionfilingnoSign": ""
    }

    data = safe_get(url, params)

    if not isinstance(data, dict):
        return []

    row_data = data.get("data", {}).get("rowData", [])
    
    if not row_data:
        return []

    if isinstance(row_data, dict):
        return convert_column_to_row(row_data)
    elif isinstance(row_data, list):
        return row_data
    
    return []


# ================= 分批存储管理器 =================
class BatchStorage:
    """
    优化版分批存储管理器
    
    特点:
    1. 按批次分组存储（每N页一个文件）
    2. 每页50条，每批100页=5000条
    3. 5万条数据 → 10个批次文件
    """
    
    def __init__(self, data_dir="variety_batch_data"):
        self.data_dir = Path(data_dir)
        self.data_dir.mkdir(exist_ok=True)
        self.pages_per_batch = CONFIG["PAGES_PER_BATCH"]
        
        # 进度文件路径
        self.progress_file = self.data_dir / "progress.json"
        
        # 加载或初始化进度
        self.progress = self._load_progress()
    
    def _get_batch_name(self, page_num):
        """根据页码获取批次文件名"""
        batch_index = (page_num - 1) // self.pages_per_batch + 1
        start_page = (batch_index - 1) * self.pages_per_batch + 1
        end_page = min(batch_index * self.pages_per_batch, CONFIG["MAX_PAGES"])
        
        # 文件名包含批次信息和页数范围
        return f"batch_{batch_index:03d}_pages_{start_page:04d}-{end_page:04d}.json"
    
    def _get_batch_index(self, page_num):
        """获取批次索引"""
        return (page_num - 1) // self.pages_per_batch + 1
    
    def _load_progress(self):
        """加载爬取进度"""
        if self.progress_file.exists():
            with open(self.progress_file, 'r', encoding='utf-8') as f:
                progress = json.load(f)
                print(f"📂 加载进度: 已爬取 {progress.get('total_records', 0):,} 条, "
                      f"最后页码 {progress.get('last_page', 0)}")
                return progress
        
        # 初始化进度
        return {
            "version": "3.0",
            "config": {
                "page_size": CONFIG["PAGE_SIZE"],
                "pages_per_batch": CONFIG["PAGES_PER_BATCH"],
                "max_pages": CONFIG["MAX_PAGES"]
            },
            "start_time": datetime.now().isoformat(),
            "last_page": 0,
            "total_records": 0,
            "crawled_pages": [],
            "batches": {}
        }
    
    def _save_progress(self):
        """保存爬取进度"""
        self.progress["last_update"] = datetime.now().isoformat()
        with open(self.progress_file, 'w', encoding='utf-8') as f:
            json.dump(self.progress, f, indent=2, ensure_ascii=False)
    
    def _load_batch_file(self, batch_file):
        """加载批次文件内容"""
        file_path = self.data_dir / batch_file
        if not file_path.exists():
            return {}
        
        with open(file_path, 'r', encoding='utf-8') as f:
            return json.load(f)
    
    def _save_batch_file(self, batch_file, data):
        """保存批次文件"""
        file_path = self.data_dir / batch_file
        with open(file_path, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
        
        file_size = file_path.stat().st_size / (1024 * 1024)  # MB
        print(f"  💾 批次文件已更新: {batch_file} ({file_size:.2f}MB)")
    
    def save_page(self, page_num, page_data):
        """
        保存单页数据到对应的批次文件
        """
        if not page_data:
            return False
        
        batch_file = self._get_batch_name(page_num)
        batch_index = self._get_batch_index(page_num)
        
        # 加载现有批次数据
        batch_data = self._load_batch_file(batch_file)
        
        # 添加新页数据（使用页码作为key）
        batch_data[str(page_num)] = page_data
        
        # 保存批次文件
        self._save_batch_file(batch_file, batch_data)
        
        # 更新进度
        if page_num not in self.progress["crawled_pages"]:
            self.progress["crawled_pages"].append(page_num)
        
        self.progress["last_page"] = page_num
        self.progress["total_records"] += len(page_data)
        
        # 更新批次信息
        if batch_index not in self.progress["batches"]:
            self.progress["batches"][batch_index] = {
                "file": batch_file,
                "pages": [],
                "records": 0,
                "start_page": (batch_index - 1) * self.pages_per_batch + 1,
                "end_page": min(batch_index * self.pages_per_batch, CONFIG["MAX_PAGES"])
            }
        
        batch_info = self.progress["batches"][batch_index]
        if page_num not in batch_info["pages"]:
            batch_info["pages"].append(page_num)
        batch_info["records"] += len(page_data)
        
        self._save_progress()
        
        # 打印保存信息
        print(f"  ✅ 第{page_num}页已存入: {batch_file}")
        print(f"  📊 本批次: {len(batch_info['pages'])}/{batch_info['end_page']-batch_info['start_page']+1}页, "
              f"{batch_info['records']:,}/{CONFIG['RECORDS_PER_BATCH']:,}条")
        print(f"  📊 全局: {self.progress['total_records']:,}/{CONFIG['TOTAL_RECORDS']:,}条 "
              f"({self.progress['total_records']/CONFIG['TOTAL_RECORDS']*100:.1f}%)")
        
        return True
    
    def is_page_crawled(self, page_num):
        """检查某页是否已爬取"""
        return page_num in self.progress["crawled_pages"]
    
    def get_next_page(self, start_page=1):
        """获取下一个需要爬取的页码"""
        for page in range(start_page, start_page + 10000):
            if not self.is_page_crawled(page):
                return page
        return None
    
    def get_statistics(self):
        """获取统计信息"""
        total_batches = len(self.progress["batches"])
        total_pages = len(self.progress["crawled_pages"])
        total_records = self.progress["total_records"]
        
        # 计算文件总大小
        total_size = 0
        for batch_info in self.progress["batches"].values():
            file_path = self.data_dir / batch_info["file"]
            if file_path.exists():
                total_size += file_path.stat().st_size
        
        completion_rate = (total_records / CONFIG["TOTAL_RECORDS"] * 100) if CONFIG["TOTAL_RECORDS"] > 0 else 0
        
        return {
            "total_batches": total_batches,
            "total_pages": total_pages,
            "total_records": total_records,
            "expected_records": CONFIG["TOTAL_RECORDS"],
            "completion_rate": completion_rate,
            "last_page": self.progress["last_page"],
            "total_size_mb": total_size / (1024 * 1024),
            "data_dir": str(self.data_dir),
            "start_time": self.progress.get("start_time"),
            "last_update": self.progress.get("last_update")
        }
    
    def get_batch_data(self, batch_index):
        """获取指定批次的所有数据（返回列表格式）"""
        batch_info = self.progress["batches"].get(batch_index)
        if not batch_info:
            return []
        
        batch_data = self._load_batch_file(batch_info["file"])
        
        # 转换为列表格式（按页码顺序）
        all_data = []
        for page_num in sorted(batch_data.keys(), key=int):
            all_data.extend(batch_data[page_num])
        
        return all_data
    
    def get_all_batches(self):
        """获取所有批次索引"""
        return sorted(self.progress["batches"].keys())
    
    def get_all_data_generator(self):
        """
        生成器：逐条返回所有数据（内存友好）
        用于导入数据库
        """
        for batch_index in self.get_all_batches():
            batch_data = self.get_batch_data(batch_index)
            for record in batch_data:
                yield record
    
    def merge_all(self, output_file="all_variety_data.json"):
        """
        合并所有数据到一个文件（仅在需要时使用）
        """
        all_data = []
        total = 0
        
        print(f"\n🔄 开始合并数据...")
        print(f"⚠️  注意: 合并 {CONFIG['TOTAL_RECORDS']:,} 条数据会占用约 {CONFIG['TOTAL_RECORDS']*0.5/1024:.1f}GB 内存")
        
        confirm = input("  确认继续？(y/n): ").strip().lower()
        if confirm != 'y':
            print("已取消")
            return None
        
        for batch_index in self.get_all_batches():
            batch_data = self.get_batch_data(batch_index)
            all_data.extend(batch_data)
            total += len(batch_data)
            print(f"  已合并批次 {batch_index:03d}: {len(batch_data):,}条, 累计{total:,}条")
        
        # 保存合并文件
        with open(output_file, 'w', encoding='utf-8') as f:
            json.dump(all_data, f, ensure_ascii=False, indent=2)
        
        file_size_mb = Path(output_file).stat().st_size / (1024 * 1024)
        print(f"\n✅ 合并完成: {total:,}条数据, {file_size_mb:.1f}MB -> {output_file}")
        
        return all_data


# ================= 分批爬取核心函数 =================
def crawl_batch(start_page=1, end_page=None):
    """
    分批爬取数据
    
    参数:
        start_page: 起始页码
        end_page: 结束页码（默认使用配置的MAX_PAGES）
    """
    if end_page is None:
        end_page = CONFIG["MAX_PAGES"]
    
    storage = BatchStorage()
    
    expected_total = (end_page - start_page + 1) * CONFIG["PAGE_SIZE"]
    expected_batches = (end_page - start_page + 1 + CONFIG["PAGES_PER_BATCH"] - 1) // CONFIG["PAGES_PER_BATCH"]
    
    print(f"\n{'='*70}")
    print(f"🚀 开始分批爬取任务")
    print(f"{'='*70}")
    print(f"  配置信息:")
    print(f"    每页数量: {CONFIG['PAGE_SIZE']} 条")
    print(f"    每批页数: {CONFIG['PAGES_PER_BATCH']} 页")
    print(f"    每批记录: {CONFIG['RECORDS_PER_BATCH']:,} 条")
    print(f"  任务范围:")
    print(f"    起始页码: {start_page}")
    print(f"    结束页码: {end_page}")
    print(f"    预计总条数: {expected_total:,} 条")
    print(f"    预计批次数: {expected_batches} 个文件")
    print(f"  其他:")
    print(f"    请求间隔: {CONFIG['REQUEST_INTERVAL']}秒")
    print(f"    存储目录: {storage.data_dir}")
    print(f"{'='*70}\n")
    
    start_time = time.time()
    
    success_pages = 0
    empty_pages = 0
    failed_pages = 0
    
    for page in range(start_page, end_page + 1):
        # 检查是否已爬取
        if storage.is_page_crawled(page):
            print(f"⏭️  第{page}页已存在，跳过")
            continue
        
        print(f"\n📄 [进度: {page}/{end_page}] 正在爬取第 {page} 页...")
        
        try:
            data = get_rowData(page=page, page_size=CONFIG["PAGE_SIZE"])
            
            if not data:
                print(f"  ⚠️ 第{page}页无数据")
                empty_pages += 1
                if empty_pages >= 3:
                    print(f"\n⚠️ 连续{empty_pages}页无数据，停止爬取")
                    break
                continue
            
            empty_pages = 0
            
            # 保存到批次文件
            storage.save_page(page, data)
            success_pages += 1
            
            time.sleep(CONFIG["REQUEST_INTERVAL"])
            
        except Exception as e:
            print(f"  ❌ 第{page}页爬取失败: {e}")
            failed_pages += 1
            time.sleep(CONFIG["REQUEST_INTERVAL"] * 3)
            continue
    
    elapsed_time = time.time() - start_time
    stats = storage.get_statistics()
    
    print(f"\n{'='*70}")
    print(f"🎉 爬取任务完成！")
    print(f"{'='*70}")
    print(f"  成功页数: {success_pages}")
    print(f"  空页数: {empty_pages}")
    print(f"  失败页数: {failed_pages}")
    print(f"  总记录数: {stats['total_records']:,} / {stats['expected_records']:,} 条")
    print(f"  完成度: {stats['completion_rate']:.1f}%")
    print(f"  总批次数: {stats['total_batches']} 个文件")
    print(f"  总文件大小: {stats['total_size_mb']:.1f} MB")
    print(f"  耗时: {elapsed_time/60:.1f} 分钟")
    print(f"  平均速度: {stats['total_records']/elapsed_time:.1f} 条/秒")
    print(f"  存储位置: {storage.data_dir}")
    print(f"{'='*70}")

    return all_data

"""

import json

def clean_value(value):
    """清理单个值，处理None和特殊字符"""
    if value is None:
        return None
    if isinstance(value, str):
        # 转义单引号，避免SQL注入
        return value.strip().replace("'", "\\'")
    return str(value)


def parse_record(record):
    """

    return (

        clean_value(record.get('cropid')),

def import_batch(records, cursor, batch_size=500):
    """
    批量导入记录
    """
    sql = """
        INSERT INTO variety_data (
            cropid, cropname, varietyname, varietyannouncementid,
            judgementno, judgementyear, judgementregion, judgementregionid,
            certificateno, breedingcompany, applycompany,
            varietycharacter, varietysource, plantrequirment,
            judgementsuggestion, outputexpression,
            isTuiguangHas, isVarietyApplyHas, isXukeByVarietynameHas,
            isXukeByApplycompanynameHas, istransgenosis, status
        ) VALUES (
            %s, %s, %s, %s,
            %s, %s, %s, %s,
            %s, %s, %s,
            %s, %s, %s,
            %s, %s,
            %s, %s, %s,
            %s, %s, %s
        )
    """
    
    inserted = 0
    for i in range(0, len(records), batch_size):
        batch = records[i:i+batch_size]
        try:
            cursor.executemany(sql, batch)
            inserted += len(batch)
            print(f"    ✅ 已入库: {inserted}/{len(records)} 条")
        except Exception as e:
            print(f"    ❌ 批次入库失败: {e}")
            # 降级：逐条插入
            for record in batch:
                try:
                    cursor.execute(sql, record)
                    inserted += 1
                except Exception as e2:
                    print(f"      ❌ 单条失败: {record[3]}, 错误: {e2}")
    
    return inserted


def load_records_from_files(data_dir, max_records=None):
    """
    从JSON文件加载记录
    max_records: 最大加载条数（用于测试）
    """
    data_path = Path(data_dir)
    batch_files = sorted(data_path.glob("batch_*.json"))
    
    if not batch_files:
        print(f"❌ 未找到JSON文件: {data_dir}")
        return []
    
    all_records = []
    
    for batch_file in batch_files:
        with open(batch_file, 'r', encoding='utf-8') as f:
            batch_data = json.load(f)
        
        # 展开所有记录
        for page_records in batch_data.values():
            for record in page_records:
                all_records.append(record)
                if max_records and len(all_records) >= max_records:
                    return all_records
    
    return all_records

            parsed_records.append(parsed)
    
    print(f"有效记录: {len(parsed_records)} 条")

    conn.close()
    
    print(f"\n✅ 导入完成: 成功 {inserted}/{len(parsed_records)} 条")
    
    return inserted

-- =====================================================
-- 1. 创建空白数据库
-- =====================================================
CREATE DATABASE IF NOT EXISTS variety_db 
CHARACTER SET utf8mb4 
COLLATE utf8mb4_unicode_ci;

USE variety_db;

-- =====================================================
-- 2. 创建作物字典表（字段来源于JSON: cropid, cropname）
-- =====================================================
CREATE TABLE crop (
    crop_id INT PRIMARY KEY COMMENT '作物ID（来自JSON cropid）',
    crop_name VARCHAR(50) NOT NULL UNIQUE COMMENT '作物名称（来自JSON cropname）',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_crop_name (crop_name)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='作物字典表';


-- =====================================================
-- 3. 创建区域字典表（字段来源于JSON: judgementregionid, judgementregion）
-- =====================================================
CREATE TABLE region (
    region_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '区域ID',
    region_code VARCHAR(20) UNIQUE COMMENT '区域代码（来自JSON judgementregionid）',
    region_name VARCHAR(100) NOT NULL COMMENT '区域名称（来自JSON judgementregion）',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_region_code (region_code),
    INDEX idx_region_name (region_name(50))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='区域字典表';


-- =====================================================
-- 4. 创建公司字典表（字段来源于JSON: breedingcompany, applycompany）
-- =====================================================
CREATE TABLE company (
    company_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '公司ID',
    company_name VARCHAR(200) NOT NULL UNIQUE COMMENT '公司名称（来自JSON breedingcompany/applycompany）',
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    INDEX idx_company_name (company_name(50))
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COMMENT='公司字典表';


-- =====================================================
-- 5. 创建品种主表（字段完全匹配JSON）
-- =====================================================
CREATE TABLE variety (
    -- 主键
    variety_id INT PRIMARY KEY AUTO_INCREMENT COMMENT '品种ID',
    varietyannouncementid VARCHAR(36) NOT NULL UNIQUE COMMENT '公告ID（来自JSON varietyannouncementid）',
    
    -- 外键关联
    crop_id INT NOT NULL COMMENT '作物ID（关联crop表）',
    region_id INT COMMENT '审定区域ID（关联region表）',
    breed_company_id INT COMMENT '育种公司ID（关联company表）',
    apply_company_id INT COMMENT '申请公司ID（关联company表）',
    
    -- 品种基本信息（直接来自JSON字段）
    varietyname VARCHAR(200) NOT NULL COMMENT '品种名称（来自JSON varietyname）',
    judgementno VARCHAR(100) COMMENT '审定编号（来自JSON judgementno）',
    judgementyear INT COMMENT '审定年份（来自JSON judgementyear）',
    certificateno VARCHAR(100) COMMENT '证书编号（来自JSON certificateno）',
    
    -- 文本内容（直接来自JSON字段）
    varietycharacter TEXT COMMENT '品种特性（来自JSON varietycharacter）',
    varietysource TEXT COMMENT '品种来源（来自JSON varietysource）',
    plantrequirment TEXT COMMENT '种植要求（来自JSON plantrequirment）',
    judgementsuggestion TEXT COMMENT '审定建议（来自JSON judgementsuggestion）',
    outputexpression TEXT COMMENT '产量表现（来自JSON outputexpression）',
    
    -- 标志字段（直接来自JSON字段）
    isTuiguangHas TINYINT DEFAULT 0 COMMENT '是否推广（来自JSON isTuiguangHas）',
    isVarietyApplyHas TINYINT DEFAULT 0 COMMENT '是否品种申请（来自JSON isVarietyApplyHas）',
    isXukeByVarietynameHas TINYINT DEFAULT 0 COMMENT '品种名是否许可（来自JSON isXukeByVarietynameHas）',
    isXukeByApplycompanynameHas TINYINT DEFAULT 0 COMMENT '申请公司是否许可（来自JSON isXukeByApplycompanynameHas）',
    istransgenosis TINYINT DEFAULT 0 COMMENT '是否转基因（来自JSON istransgenosis）',
    
    -- 状态（来自JSON status）
    status TINYINT DEFAULT 1 COMMENT '状态（来自JSON status）',
    
    -- 原始JSON中保留但未使用的字段（备查）
    judgementregionid VARCHAR(20) COMMENT '原始区域代码（备查）',
    judgementregion VARCHAR(100) COMMENT '原始区域名称（备查）',
    breedingcompany TEXT COMMENT '原始育种单位（备查）',
    applycompany TEXT COMMENT '原始申请单位（备查）',
    cropname VARCHAR(50) COMMENT '原始作物名称（备查）',
    
    -- 时间戳
    created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP,
    
    -- 外键约束
    FOREIGN KEY (crop_id) REFERENCES crop(crop_id),
    FOREIGN KEY (region_id) REFERENCES region(region_id),
    FOREIGN KEY (breed_company_id) REFERENCES company(company_id),
    FOREIGN KEY (apply_company_id) REFERENCES company(company_id),
    
    -- 索引
    INDEX idx_varietyannouncementid (varietyannouncementid),
    INDEX idx_varietyname (varietyname(50)),
    INDEX idx_judgementno (judgementno),
    INDEX idx_judgementyear (judgementyear),
    INDEX idx_crop_id (crop_id),
    INDEX idx_region_id (region_id),
    INDEX idx_breed_company (breed_company_id),
    INDEX idx_apply_company (apply_company_id),
    INDEX idx_crop_year (crop_id, judgementyear),
    INDEX idx_region_year (region_id, judgementyear),
    INDEX idx_year_status (judgementyear, status),
    
    FULLTEXT INDEX ft_varietycharacter (varietycharacter),
    FULLTEXT INDEX ft_judgementsuggestion (judgementsuggestion)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci COMMENT='品种信息主表';
